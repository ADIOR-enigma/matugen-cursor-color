# matugen-cursor-colour

</div>

This guide contains everything needed to set up a fully dynamic, color-matching, anti-aliased Wayland cursor theme that updates instantly when system colors change via Matugen hooks.

## 1. Install Dependencies
You only need `inotify-tools` to run the standalone background watcher:
```bash
sudo pacman -S inotify-tools
```
*(You also need `hyprcursor` installed, which comes with Hyprland).*

## 2. Clone the Cursor Source
Yes, **you absolutely need the source assets!** The script relies on these raw SVG templates to read, recolor on the fly, and compile into the final cursor format. Without them, the script has no shapes to build from.

```bash
mkdir -p ~/.local/share
git clone https://github.com/ndom91/rose-pine-hyprcursor ~/.local/share/hyprcursor-matugen
```

### 3. The Generator Script
Create the following script at `~/.config/hypr/scripts/matugen-cursor.sh`. This handles compiling the SVG vectors, setting legacy fallbacks, and hot-reloading Hyprland.

```bash
#!/usr/bin/env bash
# =============================================================================
# Matugen Dynamic Cursor Generator
# 
# This script reads colors generated by Matugen, recolors the raw cursor SVG 
# assets, compiles them into a Hyprcursor theme, and instantly hot-reloads 
# Hyprland and GTK to apply the new cursor natively.
# =============================================================================

set -euo pipefail

# -----------------------------------------------------------------------------
# Configuration
# -----------------------------------------------------------------------------
readonly SRC_BASE="$HOME/.local/share/hyprcursor-matugen/src"
readonly OUT_BUILD="/tmp/hyprcursor-matugen-build"
readonly COLORS_LUA="$HOME/.config/matugen/generated/hyprland-colors.lua"
readonly LOCK_FILE="/tmp/matugen-cursor.lock"
readonly LOG_FILE="/tmp/matugen-cursor.log"

# Prevent concurrent executions which could corrupt the build directory
exec 9>"$LOCK_FILE"
if ! flock -w 60 9; then 
  echo "[matugen-cursor] Lock timeout, skipping execution." >> "$LOG_FILE" 2>&1
  exit 0
fi

# -----------------------------------------------------------------------------
# Utility Functions
# -----------------------------------------------------------------------------
log() { 
  echo "[matugen-cursor] $(date '+%Y-%m-%d %H:%M:%S') - $*" >> "$LOG_FILE" 2>&1
}

die() { 
  log "CRITICAL ERROR: $*"
  exit 1
}

# Extracts the current system cursor size from GNOME settings
fetch_cursor_size() {
  local size
  size=$(gsettings get org.gnome.desktop.interface cursor-size 2>/dev/null | grep -oE '[0-9]+' | head -n1 || true)
  [[ -z "$size" || ! "$size" =~ ^[0-9]+$ ]] && size="18"
  
  # Clamp the size to safe boundaries
  echo $(( size < 16 ? 16 : (size > 128 ? 128 : size) ))
}

# Ensures the colors file is fully written before proceeding
wait_for_colors() {
  for _ in {1..5}; do
    if [[ -s "$COLORS_LUA" ]] && grep -q "primary" "$COLORS_LUA" 2>/dev/null; then 
      return 0
    fi
    sleep 0.2
  done
  die "Colors file missing or empty: $COLORS_LUA"
}

# Extracts a specific color from the Matugen generated Lua file
get_color() {
  local key="$1" fallback="$2"
  local raw
  raw=$(grep -E "^\s*$key\s*=" "$COLORS_LUA" 2>/dev/null | head -n1 | grep -oE '[0-9a-fA-F]{6,8}' | head -n1 | cut -c1-6 || true)
  [[ -n "$raw" ]] && echo "#$raw" || echo "$fallback"
}

# -----------------------------------------------------------------------------
# Core Build Logic
# -----------------------------------------------------------------------------

# Recolors all raw SVG templates using the active system colors
build_cursors() {
  log "Building raw SVG assets..."
  
  # Convert hex colors to uppercase for sed replacement matching
  local fill_up outline_up
  fill_up=$(echo "$PRIMARY" | tr 'a-f' 'A-F')
  outline_up=$(echo "$BACKGROUND" | tr 'a-f' 'A-F')

  # Clean previous temporary builds
  rm -rf "$OUT_BUILD" "/tmp/theme_matugen-cursor-"*
  mkdir -p "$OUT_BUILD/cursors"
  
  # Generate the manifest required by hyprcursor
  cat > "$OUT_BUILD/manifest.hl" <<EOF
name = $THEME_NAME
description = Matugen generated cursor
version = 1.0
cursors_directory = cursors
EOF

  # Iterate through all available cursor shapes
  for cursor_dir in "$SRC_BASE"/*; do
    [[ -d "$cursor_dir" ]] || continue
    local name
    name=$(basename "$cursor_dir")
    
    mkdir -p "$OUT_BUILD/cursors/$name"
    cp "$cursor_dir/meta.hl" "$OUT_BUILD/cursors/$name/meta.hl" 2>/dev/null || true
    
    # Improve anti-aliasing for the cursors
    sed -i "s/resize_algorithm = none/resize_algorithm = bilinear/g" "$OUT_BUILD/cursors/$name/meta.hl" 2>/dev/null || true
    
    # Recolor each SVG component in the shape directory
    for svg in "$cursor_dir"/*.svg; do
      [[ -f "$svg" ]] || continue
      local bname
      bname=$(basename "$svg")
      local tmp_svg
      tmp_svg=$(mktemp)
      
      # The base template uses #E0DEF4 for fill and #191724 for outline
      if sed -e "s/#E0DEF4/$fill_up/gI" -e "s/#191724/$outline_up/gI" "$svg" > "$tmp_svg"; then
        mv "$tmp_svg" "$OUT_BUILD/cursors/$name/$bname"
      else
        rm -f "$tmp_svg"
      fi
    done
  done
}

# Compiles the recolored SVGs into native .hlc binaries
compile_theme() {
  log "Compiling .hlc binaries via hyprcursor-util..."
  local build_out="/tmp/theme_$THEME_NAME"
  rm -rf "$build_out"
  
  if ! timeout 30 hyprcursor-util --create "$OUT_BUILD" >> "$LOG_FILE" 2>&1; then
    die "hyprcursor-util compilation failed. Check logs."
  fi
}

# Installs the compiled theme to the user's icon directory with legacy fallbacks
install_theme() {
  log "Installing theme to ~/.local/share/icons..."
  local icon_dest="$HOME/.local/share/icons/$THEME_NAME"
  local build_out="/tmp/theme_$THEME_NAME"
  
  # Atomic replacement of the theme directory
  rm -rf "$icon_dest" "$icon_dest.tmp"
  cp -r "$build_out" "$icon_dest.tmp"
  mv "$icon_dest.tmp" "$icon_dest"
  
  cp "$OUT_BUILD/manifest.hl" "$icon_dest/manifest.hl" 2>/dev/null || true

  # Ensure GTK/XWayland apps gracefully fall back to Adwaita instead of crashing
  cat > "$icon_dest/index.theme" <<EOF
[Icon Theme]
Name=$THEME_NAME
Comment=Matugen generated dynamic cursor
Inherits=Adwaita,hicolor
EOF

  # Set the system-wide default inheritance
  mkdir -p "$HOME/.local/share/icons/default" "$HOME/.icons/default"
  echo -e "[Icon Theme]\nInherits=$THEME_NAME,Adwaita,hicolor" > "$HOME/.local/share/icons/default/index.theme"
  echo -e "[Icon Theme]\nInherits=$THEME_NAME,Adwaita,hicolor" > "$HOME/.icons/default/index.theme"
}

# Applies the new theme instantly to the live environment
apply_live() {
  log "Triggering live environment hot-reloads..."
  
  # Tell GTK applications about the new theme
  gsettings set org.gnome.desktop.interface cursor-theme "$THEME_NAME" 2>/dev/null || true
  
  # Export clean Lua environment variables for Hyprland to source
  mkdir -p "$HOME/.config/hypr/generated"
  cat > "$HOME/.config/hypr/generated/cursor.lua" <<EOF
hl.env("XCURSOR_THEME", "$THEME_NAME")
hl.env("HYPRCURSOR_THEME", "$THEME_NAME")
hl.env("XCURSOR_SIZE", "$SIZE")
hl.env("HYPRCURSOR_SIZE", "$SIZE")
EOF

  # Force Hyprland to instantly adopt the new cursor
  timeout 5 hyprctl setcursor "$THEME_NAME" "$SIZE" >> "$LOG_FILE" 2>&1 || true
  timeout 5 hyprctl reload >> "$LOG_FILE" 2>&1 || true
  
  log "Cursor update complete."
}

# -----------------------------------------------------------------------------
# Main Execution
# -----------------------------------------------------------------------------
main() {
  [[ -d "$SRC_BASE" ]] || die "Cursor source base missing: $SRC_BASE"
  
  readonly SIZE="$(fetch_cursor_size)"
  wait_for_colors
  
  readonly PRIMARY="$(get_color "primary" "#b2d189")"
  readonly BACKGROUND="$(get_color "background" "#12140e")"
  
  # We append the hex code as a cache-busting suffix. 
  # This forces Hyprland to load the new cursor from disk instantly.
  readonly THEME_NAME="matugen-cursor-${PRIMARY:1}"
  
  log "--- Starting Build ---"
  log "Primary: $PRIMARY | Background: $BACKGROUND | Size: $SIZE"
  
  build_cursors
  compile_theme
  install_theme
  apply_live
}

main
```

## 4. The Watcher Daemon
Create the following script at `~/.config/hypr/scripts/matugen-cursor-watcher.sh`.

```bash
#!/usr/bin/env bash
# =============================================================================
# Matugen Cursor Watcher Daemon
#
# This script runs perpetually in the background. It uses inotifywait to 
# monitor the Lua file that Matugen generates. When Matugen updates the file 
# with new colors, this daemon instantly triggers the cursor compiler script.
# =============================================================================

set -euo pipefail

# -----------------------------------------------------------------------------
# Configuration
# -----------------------------------------------------------------------------
readonly WATCH_DIR="$HOME/.config/matugen/generated"
readonly WATCH_FILE="hyprland-colors.lua"
readonly BUILD_SCRIPT="$HOME/.config/hypr/scripts/matugen-cursor.sh"
readonly LOG_FILE="/tmp/matugen-cursor.log"

# -----------------------------------------------------------------------------
# Utility Functions
# -----------------------------------------------------------------------------
log() { 
  echo "[watcher] $(date '+%Y-%m-%d %H:%M:%S') - $*" >> "$LOG_FILE"
}

cleanup() { 
  log "Shutting down the watcher daemon safely."
  exit 0
}

# Trap termination signals so the script exits cleanly
trap cleanup SIGINT SIGTERM EXIT

# -----------------------------------------------------------------------------
# Main Loop
# -----------------------------------------------------------------------------
main() {
  log "--- Starting Watcher Daemon ---"
  
  # Ensure the directory exists so inotifywait doesn't fail
  mkdir -p "$WATCH_DIR"
  
  # Verify inotifywait is installed
  if ! command -v inotifywait >/dev/null 2>&1; then
    log "CRITICAL ERROR: 'inotifywait' is not installed. Please install 'inotify-tools'."
    exit 1
  fi

  # Monitor the directory for close_write, moved_to, and create events.
  # This ensures we catch Matugen updates whether it writes in-place or uses atomic temp file replacement.
  inotifywait -m -e close_write -e moved_to -e create --format '%f' "$WATCH_DIR" 2>/dev/null | \
  while read -r file; do
    if [[ "$file" == "$WATCH_FILE" ]]; then
      log "Detected a color update in $WATCH_FILE! Triggering cursor compilation..."
      # Spawn the build script in the background so we immediately resume watching
      bash "$BUILD_SCRIPT" &
    fi
  done
}

main
```
Make both scripts executable:
```bash
chmod +x ~/.config/hypr/scripts/matugen-cursor*.sh
```
Finally, tell Hyprland to launch the watcher daemon when you log in:

**For standard Hyprland (`hyprland.conf`):**
```bash
exec-once = bash ~/.config/hypr/scripts/matugen-cursor-watcher.sh
```

**For Lua-based Hyprland wrappers (`autostart.lua`):**
```lua
hl.on("hyprland.start", function()
    hl.exec_cmd("bash ~/.config/hypr/scripts/matugen-cursor-watcher.sh")
end)
```

## 5. Configuration & Anti-Aliasing

To avoid pixelation in the `hypr-dynamic-cursors` plugin, ensure it is set to **not** force nearest-neighbor rendering:
```lua
-- In your plugins config:
dynamic_cursors = {
    enabled = true,
    mode = "stretch",
    hyprcursor = {
        enabled = true,
        nearest = false, -- Critical for avoiding pixelation!
        resolution = 128,
    },
}
```

To dynamically load the theme without slowing down Hyprland boot, source the Lua file generated by the cursor script!
```lua
-- In your environment_variables.lua:
-- Matugen cursor: load cached environment variables generated by the script
-- This avoids slow gsettings shell commands during boot
pcall(dofile, os.getenv("HOME") .. "/.config/hypr/generated/cursor.lua")
```
